function renderEquipmentCards() { const container = document.getElementById('equipment-grid'); container.innerHTML = ''; filteredData.forEach(model => { const specs = model.specs || {}; const subcategory = model.subcategory || ''; const isElectric = subcategory.toLowerCase().includes('electric'); const isEngine = subcategory.toLowerCase().includes('engine'); const isHybrid = subcategory.toLowerCase().includes('hybrid'); const isTelehandler = subcategory.includes('Telehandler'); // Determine power badge let powerBadge = ''; if (isElectric) { powerBadge = ` Electric`; } else if (isHybrid) { powerBadge = ` Hybrid`; } else if (isEngine) { powerBadge = ` Engine`; } let weight = specs.operating_weight_lbs || 0; let platformHeight = specs.platform_height_elevated_ft || specs.platform_height_ft || 0; let capacity = specs.platform_capacity_lbs || specs.rated_capacity_lbs || 0; let liftHeight = specs.max_lift_height_ft || 0; if (currentUnit === 'metric') { weight = Math.round(weight * 0.4536); platformHeight = platformHeight > 0 ? (platformHeight * 0.3048).toFixed(1) : 0; capacity = Math.round(capacity * 0.4536); liftHeight = liftHeight > 0 ? (liftHeight * 0.3048).toFixed(1) : 0; } const card = document.createElement('div'); card.className = `equipment-card bg-slate-900 hover:bg-slate-800 border border-slate-800 rounded-3xl p-6 cursor-pointer`; card.onclick = () => showModelDetail(model.id); let statsHTML = ''; if (isTelehandler) { statsHTML = `
Max Lift Height
${liftHeight}
Capacity
${capacity}
Weight
${weight}
`; } else { statsHTML = `
Platform Height
${platformHeight}
Capacity
${capacity}
Weight
${weight}
`; } card.innerHTML = `
${subcategory} ${powerBadge}
${model.brand} ${model.model}
${statsHTML}
Click for full specs →
`; container.appendChild(card); }); }
